def require(ok, message):
    if not ok:
        raise AssertionError(message)

def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda:stream.read(1048576), b''):
            h.update(block)
    return h.hexdigest()

def read(path):
    return json.loads(Path(path).read_text(), parse_constant=lambda x:(_ for _ in ()).throw(ValueError(x)))

def positive_count(n):
    require(type(n) is int and n > 0, 'Nonpositive/noninteger chain count')

def chain(text):
    parts = text.split(',')
    require(len(parts)==4 and all(v in '012345' and len(v)==1 for v in parts), 'Malformed chain')
    g,e,r,f = map(int, parts)
    # Frozen V3 has scripted_explore, disabled pt_walk/pt_hold, and only pt_sap late.
    require(g!=5 or e==5, 'SAP exploration must be skipped')
    require(g==5 or e!=5, 'Exploration cannot create SAP')
    require(e==r, 'Disabled general rule changed action')
    require(r==f or (r,f)==(0,5), 'Unsupported late transition')
    return g,e,r,f

def rebuild(summary):
    scalars = {k:0 for k in ('active_decisions','forced_active_decisions','choice_active_decisions',
        'choice_greedy_final_equal','negative_masked_decisions','masked_decisions','greedy_final_equal',
        'any_stage_change','changes_explore','changes_rule','changes_late','changed_then_final_greedy',
        'explore_reverted','rule_reverted')}
    tables = {k:Counter() for k in ('choice_by_final_action','by_greedy_action','by_final_action','chains')}
    for entry,n in summary['legal_count_chains'].items():
        positive_count(n)
        pieces = entry.split('|')
        require(len(pieces)==2 and pieces[0] in ('1','2','3','4','5','6'), 'Invalid legal count')
        count = int(pieces[0]); g,e,r,f = chain(pieces[1])
        require(len({g,e,r,f})<=count, 'More observed classes than legal actions')
        if count==1:
            require(g==e==r==f==0, 'Single legal action must be STAY')
        tests = dict(active_decisions=True,forced_active_decisions=count==1,choice_active_decisions=count>=2,
            choice_greedy_final_equal=count>=2 and g==f,greedy_final_equal=g==f,
            any_stage_change=not(g==e==r==f),changes_explore=g!=e,changes_rule=e!=r,changes_late=r!=f,
            changed_then_final_greedy=g==f and not(g==e==r==f),
            explore_reverted=g!=e and (r==g or f==g),rule_reverted=e!=r and e==f)
        for k,v in tests.items():
            scalars[k] += n if v else 0
        tables['chains'][pieces[1]] += n
        tables['by_greedy_action'][str(g)] += n
        tables['by_final_action'][str(f)] += n
        if count>=2:
            tables['choice_by_final_action'][str(f)] += n
    for text,n in summary['negative_chains'].items():
        positive_count(n); chain(text)
        require(text=='0,0,0,0', 'Negative-energy decision must remain STAY in fixed V3')
        scalars['negative_masked_decisions'] += n
    scalars['masked_decisions'] = scalars['active_decisions']+scalars['negative_masked_decisions']
    result = {**scalars, **{k:dict(v) for k,v in tables.items()},
              'legal_count_chains':dict(summary['legal_count_chains']),
              'negative_chains':dict(summary['negative_chains'])}
    require(result==summary, 'Summary counter differs from chain-derived count')
    require(result['forced_active_decisions']+result['choice_active_decisions']==result['active_decisions'], 'Choice partition')
    return result

def aggregate(items):
    result = {}
    for item in items:
        for k,v in item.items():
            if isinstance(v,dict):
                bucket=result.setdefault(k,{})
                for key,n in v.items():
                    bucket[key]=bucket.get(key,0)+n
            else:
                require(type(v) is int and v>=0, 'Invalid aggregate counter')
                result[k]=result.get(k,0)+v
    return result

def check_reference(actual, reference):
    require(set(actual)==set(reference), 'Reference field set differs')
    for k,v in reference.items():
        if k=='secs':
            require(math.isfinite(float(actual[k])), 'Nonfinite duration')
        elif k in ('side','opponent'):
            require(actual[k]==v, 'Reference string differs: '+k)
        else:
            x,y=float(actual[k]),float(v)
            require(math.isfinite(x) and math.isfinite(y) and abs(x-y)<=1e-9, 'Reference numeric differs: '+k)

def current_head():
    git = ROOT/'.git'
    require(git.is_dir(), 'Unexpected repository layout')
    value=(git/'HEAD').read_text().strip()
    if not value.startswith('ref: '):
        return value
    ref=value[5:]
    if (git/ref).is_file():
        return (git/ref).read_text().strip()
    for line in (git/'packed-refs').read_text().splitlines():
        if line and not line.startswith(('#','^')):
            h,name=line.split(' ',1)
            if name==ref:
                return h
    raise AssertionError('Unresolved HEAD')
