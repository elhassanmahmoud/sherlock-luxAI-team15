from pathlib import Path
import json,csv,collections,hashlib,math
R=Path.cwd();M=R/'experiments/candidates_20261020.json';m=json.loads(M.read_text());data=[]

 run=Path(source).parent.parent;manifest=R/run/'manifest.json';log=R/run/'train.csv';f=json.loads(manifest.read_text())

assert len(data)==18 and sum(x['completed_rows'] for x in data)==66000
print(json.dumps({'complete':True,'scope':'JSON/CSV provenance audit; individual source checkpoint files unavailable locally, so their tensor identity/training-step choice is not newly verified.
