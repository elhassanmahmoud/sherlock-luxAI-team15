import ast
import hashlib
from pathlib import Path
import sys
import numpy as np

raw=SOURCE.read_bytes()
names={'require','bootstrap','fresh_comparison','fresh_gate','direct_stat','direct_gate'}
nodes=[n for n in ast.parse(raw).body if isinstance(n,ast.FunctionDef) and n.name in names]
ns={}
exec(compile(ast.Module(body=nodes,type_ignores=[]),str(SOURCE),'exec'),ns)
reference={};candidate={}
for i,seed in enumerate(range(510200,510400)):
    for j,side in enumerate(('player_0','player_1')):
        for opp in ('default','solution92'):

ci=np.percentile(map_delta[ix].mean(1),[2.5,97.5])
assert np.array_equal(result['solution92']['ci95'],ci)
for record in result.values():
    assert record['maps']==200 and record['games_per_arm']==400
    assert record['bootstrap_samples']==5000 and record['bootstrap_seed']==0
    assert abs(record['half_units']/800-record['delta'])<1e-15
assert result['solution92']['half_units']==80
assert result['solution92']['reference_mean']==.5 and result['solution92']['candidate_mean']==.6
assert result['default']['reference_mean']==.5 and result['default']['candidate_mean']==.5
for record in result.values():
    assert abs(record['candidate_mean']-record['reference_mean']-record['delta'])<1e-15
assert ns['fresh_gate'](result)['passed']
result['default']['half_units']=-16

opp='self:/frozen/reference.pth'
