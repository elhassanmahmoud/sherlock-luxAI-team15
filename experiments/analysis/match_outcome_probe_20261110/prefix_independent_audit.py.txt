def sha(path):
 h=hashlib.sha256()
 with open(path,'rb') as f:
  for chunk in iter(lambda:f.read(1024*1024),b''):h.update(chunk)
 return h.hexdigest()

def exact(value):
 h=hashlib.sha256()
 def emit(b):h.update(struct.pack('!Q',len(b)));h.update(b)
 def visit(o):
  if isinstance(o,dict):
   emit(b'dict')
   for k in sorted(o,key=lambda k:(type(k).__name__,str(k))):visit(k);visit(o[k])
  elif isinstance(o,(list,tuple)):
   emit(type(o).__name__.encode());emit(str(len(o)).encode())
   for x in o:visit(x)
  elif o is None or type(o) in (str,bool,int,float):
   emit(type(o).__name__.encode());emit(repr(o).encode())
  else:raise TypeError(type(o))
 visit(value);return h.hexdigest()

def action_sha(actions):
 check(set(actions)=={'player_0','player_1'},'Exactly two teams','actions')
 data=bytearray()
 for side in sorted(actions):
  a=actions[side]
  check(len(a)==16 and all(len(row)==3 and all(type(x) is int for x in row) for row in a),'Integer 16x3 actions','actions')
  check(all(0<=row[0]<=5 for row in a),'Action IDs in engine range','actions')
  header=json.dumps(dict(side=side,shape=[16,3],dtype='<i8'),sort_keys=True).encode()
  raw=struct.pack('<48q',*(x for row in a for x in row))
  data.extend(struct.pack('!Q',len(header)));data.extend(header)
  data.extend(struct.pack('!Q',len(raw)));data.extend(raw)
 return hashlib.sha256(data).hexdigest()

def read_rows(path):

def inspect_frames(frames):
 check(len(frames) in (101,202,303),'A complete current-match prefix length')
 check([f['step'] for f in frames]==list(range(len(frames))),'Original-prefix global clock')
 points=[0,0];wins=[0,0];terminals=[]
 for i,f in enumerate(frames):
  check(exact({k:v for k,v in f.items() if k!='sha256'})==f['sha256'],'Frame typed hash','hashes')
  check(action_sha(f['actions'])==f['action_sha256'],'Action raw-byte hash','hashes')
  check(all(type(x) is int and x>=0 for x in f['points']) and len(f['points'])==2,'Nonnegative point increments','labels')
  points=[points[t]+f['points'][t] for t in (0,1)]
  check(points==f['point_state'],'Point balance including every final action','labels')
  if i:
   check(f['before_state']==frames[i-1]['after_state'],'State continuity','frames')
   check(f['before_obs']==frames[i-1]['after_obs'],'Observation continuity','frames')
  term=f['terminal']
  check(bool(term)==(i%101==100),'Match endpoint clock','labels')
  check(f['done']==(i==302),'Episode endpoint clock','labels')
  if term:
   check(term['points']==points,'Terminal points','labels')
   check(len(term['wins'])==2 and all(type(x) is int for x in term['wins']),'Integer terminal wins','labels')
   delta=[term['wins'][t]-wins[t] for t in (0,1)]
   check(delta in ([1,0],[0,1]),'Exactly one winner from wins increment','labels')
   winner=delta.index(1)
   check(term['winner']==winner,'Recorded winner matches actual wins increment','labels')
   energy=term['energies']
   check(len(energy)==2 and all(type(x) is int and math.isfinite(x) for x in energy),'Recorded terminal energy shape','labels')
   reason='points' if points[0]!=points[1] else 'energy' if energy[0]!=energy[1] else 'random'
   check(term['reason']==reason,'Point, energy, random priority','labels')
   if reason!='random':
    decisive=points if reason=='points' else energy
    check(winner==int(decisive[1]>decisive[0]),'Winner agrees with decisive total','labels')
   terminals.append(term);wins=term['wins'][:];points=[0,0]
 return terminals

def execution(runs):
 return dict(entered_runs=len(runs),completed_runs=sum(r['status']=='complete' for r in runs),
  failed_runs=sum(r['status']=='failed' for r in runs),
  entered_planned_full_length_runs=sum(r['planned_steps']==303 for r in runs),
  returned_full_length_runs=sum(r['steps']==303 for r in runs),
  observed_episode_ends=sum(bool(r['ended_episode']) for r in runs),
  completed_short_runs=sum(r['status']=='complete' and r['steps']<303 for r in runs),
  incomplete_runs=sum(r['status']!='complete' for r in runs),
  attempted_engine_steps=sum(r['attempted_steps'] for r in runs),
  returned_engine_steps=sum(r['steps'] for r in runs),archived_frames=sum(r['archived_frames'] for r in runs))

def summary(rows):
 eff=[r for r in rows if r['effective']];pos=[r for r in rows if r['delta_win']>0]
 b=len(eff)>=24 and len({r['seed'] for r in eff})>=6
 c=len(pos)>=3 and len({r['seed'] for r in pos})>=3
 return dict(cases=len(rows),effective_pairs=len(eff),effective_maps=len({r['seed'] for r in eff}),
  positive_flips=len(pos),positive_maps=len({r['seed'] for r in pos}),
  negative_flips=sum(r['delta_win']<0 for r in rows),zero_flips=sum(r['delta_win']==0 for r in rows),
  delta_points_sum=sum(r['delta_points'] for r in rows),
  h12_end_sign_reversals=sum(r['delta_h12']*r['delta_points']<0 for r in rows),
  influence_gate=b,label_gate=c,passed=b and c)
