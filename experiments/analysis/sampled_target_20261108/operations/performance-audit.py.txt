def need(condition, message):
    if not condition:
        raise ValueError(message)


def digest(path):
    result = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            result.update(block)
    return result.hexdigest()


def unique(pairs):
    result = {}
    for key, value in pairs:
        need(key not in result, f"Duplicate JSON key: {key}")
        result[key] = value
    return result


def read(path):
    def bad(value):
        raise ValueError(f"Nonfinite JSON value: {value}")
    return json.loads(Path(path).read_text(), object_pairs_hook=unique, parse_constant=bad)


def sha(value):
    need(isinstance(value, str) and re.fullmatch("[0-9a-f]{64}", value), "Invalid SHA256")
    return value


class Evidence:
    def __init__(self):
        self.files = {}

    def bind(self, path, expected):
        path = Path(path)
        need(path.is_absolute() and ".." not in path.parts, "Evidence path must be absolute and normalized")
        path = path.resolve(strict=True)
        need(path.is_file() and digest(path) == sha(expected), f"Changed evidence: {path}")
        need(str(path) not in self.files or self.files[str(path)] == expected, f"Conflicting evidence: {path}")
        self.files[str(path)] = expected
        return path

    def mapping(self, values):
        need(isinstance(values, dict) and values, "Missing evidence inventory")
        for path, value in values.items():
            self.bind(path, value)

    def json(self, path, value):
        return read(self.bind(path, value))

    def finish(self):
        for path, value in self.files.items():
            need(digest(path) == value, f"Evidence changed during audit: {path}")


def load_support(path):
    global H, np, torch
    need(digest(path) == SUPPORT_SHA, "Independent typed-hash support differs")
    spec = importlib.util.spec_from_file_location("independent_sampled_performance_support", path)
    H = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(H)
    np, torch = H.np, H.torch
    torch.set_num_threads(1)
    torch.set_num_interop_threads(1)

        need(entry == dict(path=relative, sha256=digest(root / relative), source_checkpoint_sha256=original_entry["checkpoint_sha256"],

def csv_rows(path, job):
    with path.open(newline="") as stream:
        reader = csv.DictReader(stream)
        need(reader.fieldnames is not None and len(reader.fieldnames) == len(FIELDS) and set(reader.fieldnames) == set(FIELDS), "Unexpected CSV schema")
        rows = list(reader)
    need(len(rows) == 40, "CSV must contain exactly 40 rows")
    expected = {(seed, side) for seed in job["seeds"] for side in SIDES}; seen = set(); result = []
    for row in rows:
        need(None not in row and all(value not in (None, "") for value in row.values()), "Missing CSV field")
        values = {key: float(value) for key, value in row.items() if key not in ("side", "opponent")}
        need(all(math.isfinite(value) for value in values.values()) and values["seed"].is_integer(), "Nonfinite/noninteger CSV data")
        key = (int(values["seed"]), row["side"])
        need(key in expected and key not in seen and row["opponent"] == job["opponent"], "Unexpected or duplicated CSV game")
        seen.add(key)
        need(values["steps"] == 303 and values["point_counting_version"] == 2 and row["point_counting_version"] == "2" and
             values["secs"] >= 0, "Incomplete game/counting/runtime")
        for name in ("our_wins", "opp_wins", "our_points", "opp_points", "sap_attempts", "sap_hits", "relics_found", "relics_seen",
                     "known_point_tiles", "belief_contradictions", "unit_steps"):
            need(values[name].is_integer() and values[name] >= 0, f"Invalid nonnegative integer {name}")
        for name in ("wait_frac", "invalid_frac", "greedy_masked_frac", "stack_frac"):
            need(0 <= values[name] <= 1, f"Invalid fraction {name}")
        need(values["our_wins"] + values["opp_wins"] <= 3, "Invalid total match wins")
        score = 1.0 if values["our_wins"] > values["opp_wins"] else 0.0 if values["our_wins"] < values["opp_wins"] else 0.5
        point = 1.0 if values["our_points"] > values["opp_points"] else 0.0 if values["our_points"] < values["opp_points"] else 0.5
        need(values["result"] == score and values["game_win"] == int(score == 1) and values["point_win"] == point,
             "CSV scores disagree with recorded match/point totals")
        result.append(dict(seed=key[0], side=key[1], opponent=row["opponent"], result=score, half=int(2 * score)))
    need(seen == expected, "Missing CSV keys")
    return result


def counts(rows):
    wins = sum(row["half"] == 2 for row in rows); draws = sum(row["half"] == 1 for row in rows)
    size = len(rows); half = 2 * wins + draws
    return dict(games=size, wins=wins, draws=draws, losses=size - wins - draws, half_wins=half,
                score=half / (2 * size), strict_winrate=wins / size)


def interval(values):
    need(len(values) == 20 and all(math.isfinite(value) for value in values), "Bootstrap requires 20 complete map clusters")
    # Independently generate map INDICES; a chosen index keeps all sides/opponents/starts together.
    rng = np.random.Generator(np.random.PCG64(0))
    indices = rng.integers(0, 20, size=(5000, 20))
    samples = np.asarray(values, dtype=np.float64)[indices].mean(axis=1)
    ordered = sorted(float(value) for value in samples)
    def percentile(fraction):
        position = (len(ordered) - 1) * fraction
        lower = math.floor(position); upper = math.ceil(position)
        return ordered[lower] + (ordered[upper] - ordered[lower]) * (position - lower)
    return percentile(.025), percentile(.975)


def clusters(rows, first, width):
    values = []
    for seed in range(first, first + 20):
        group = [row for row in rows if row["seed"] == seed]
        need(len(group) == width, "Incomplete map cluster")
        values.append(sum(row["half"] for row in group) / (2 * width))
    return values


def comparison(candidate, reference, opponent=None):
    width = 2 if opponent is not None else 4
    cv = clusters(candidate, 9930000, width); rv = clusters(reference, 9930000, width)
    deltas = [left - right for left, right in zip(cv, rv)]
    low, high = interval(deltas)
    result = dict(maps=20, score=sum(cv) / 20, ref_score=sum(rv) / 20,
                  delta=sum(deltas) / 20, low=low, high=high,
                  candidate=counts(candidate), reference=counts(reference))
    if opponent is not None:
        result.update(opponent=opponent, games=40, excluded_candidate=0, excluded_reference=0)
    return result, deltas


def statistics(rows):
    starts = {}; gains = []; direct_halves = []; panel_vectors = []; direct_vectors = []
    for seed in (45, 46):
        control = rows[f"control{seed}_panel_0"] + rows[f"control{seed}_panel_1"]
        sampled = rows[f"sampled{seed}_panel_0"] + rows[f"sampled{seed}_panel_1"]
        duel = rows[f"sampled{seed}_direct"]
        panel, vector = comparison(sampled, control); panel_vectors.append(vector)
        opponents = {opponent: comparison(rows[f"sampled{seed}_panel_{index}"], rows[f"control{seed}_panel_{index}"], opponent)[0]
                     for index, opponent in enumerate(OPPONENTS)}
        dv = clusters(duel, 9930100, 2); direct_vectors.append(dv); low, high = interval(dv)
        direct = dict(maps=20, games=40, score=sum(dv) / 20, low=low, high=high, candidate=counts(duel))
        gain = panel["candidate"]["half_wins"] - panel["reference"]["half_wins"]
        gains.append(gain); direct_halves.append(direct["candidate"]["half_wins"])
        starts[str(seed)] = dict(panel=panel, opponents=opponents, direct=direct, panel_half_win_gain=gain)
    panel_mean = [(a + b) / 2 for a, b in zip(*panel_vectors)]
    direct_mean = [(a + b) / 2 for a, b in zip(*direct_vectors)]
    low, high = interval(panel_mean); dl, dh = interval(direct_mean)
    checks = dict(start45_panel_positive=gains[0] > 0, start46_panel_positive=gains[1] > 0,
                  mean_panel_gain_at_least5pp=sum(gains) >= 16, start45_direct_at_least50pct=direct_halves[0] >= 40,
                  start46_direct_at_least50pct=direct_halves[1] >= 40, combined_direct_above50pct=sum(direct_halves) > 80)
    return dict(games=400, maps_per_cohort=20, starts=starts, panel_half_win_gain_sum=sum(gains), direct_half_win_sum=sum(direct_halves),
                checks=checks, passed=all(checks.values()), promoted=False,
                descriptive_mean_panel=dict(delta=sum(panel_mean) / 20, low=low, high=high, maps=20, fixed_starts=2),
                descriptive_mean_direct=dict(score=sum(direct_mean) / 20, low=dl, high=dh, maps=20, fixed_starts=2))


def equal_numbers(actual, expected, path="summary"):
    if isinstance(expected, dict):
        need(isinstance(actual, dict) and set(actual) == set(expected), f"Different fields at {path}")
        for key in expected:
            equal_numbers(actual[key], expected[key], path + "." + key)
    elif isinstance(expected, bool) or isinstance(expected, int):
        need(type(actual) is type(expected) and actual == expected, f"Different exact value at {path}")
    elif isinstance(expected, float):
        need(type(actual) in (float, int) and math.isfinite(actual) and math.isclose(actual, expected, rel_tol=1e-12, abs_tol=1e-12),
             f"Different numerical value at {path}: {actual!r} vs {expected!r}")
    else:
        need(type(actual) is type(expected) and actual == expected, f"Different value at {path}")

    need(set(state["protected"]) == {str(root / "Gruppe15.zip"), *(str(root / f"agents/sherlock/weights/agent_p{i}.pth") for i in (0, 1))} and
         all(accepted["protected"][path] == value for path, value in state["protected"].items()) and
         all(state["protected"][str(root / f"agents/sherlock/weights/agent_p{i}.pth")] == V14_SHA for i in (0, 1)), "V14/ZIP protection differs")
    provenance_path = output / "provenance.json"; frozen = ledger.json(provenance_path, state["inputs"][str(provenance_path)])

         study_kind="fixed_submitted_successor_method_pilot", inference_config=admission["inference_config"]), "Frozen provenance differs")
