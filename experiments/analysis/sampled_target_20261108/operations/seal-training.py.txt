def need(ok, message):
    if not ok:
        raise ValueError(message)


def sha(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for chunk in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def hex64(value):
    return isinstance(value, str) and re.fullmatch("[0-9a-f]{64}", value) is not None


def absolute(value):
    p = Path(value)
    need(p.is_absolute() and ".." not in p.parts, "Expected absolute non-traversing path")
    return p


def read(path):
    def pairs(items):
        result = {}
        for key, value in items:
            need(key not in result, "Duplicate JSON key")
            result[key] = value
        return result
    def nonfinite(value):
        raise ValueError("Nonfinite JSON constant: " + value)
    return json.loads(Path(path).read_text(), object_pairs_hook=pairs, parse_constant=nonfinite)


def publish(path, data):
    """Complete bytes first; publish once, atomically, without replacing anything."""
    need(not path.exists() and path.parent.is_dir(), "Output exists or parent missing")
    if not isinstance(data, bytes):
        data = (json.dumps(data, indent=2, allow_nan=False) + "\n").encode()
    temporary = None
    try:
        with tempfile.NamedTemporaryFile(dir=path.parent, prefix=".seal-", delete=False) as stream:
            temporary = Path(stream.name)
            stream.write(data); stream.flush(); os.fsync(stream.fileno())
        os.link(temporary, path)
    finally:
        if temporary is not None:
            temporary.unlink(missing_ok=True)


def verify(mapping):
    need(isinstance(mapping, dict) and bool(mapping), "Empty hash inventory")
    for path, expected in mapping.items():
        need(hex64(expected) and sha(absolute(path)) == expected, "Changed input: " + path)


def closed_processes(state):
    """Read process table only; do not kill, wait for, or restart any process."""
    owner = state["owner_pid"]; wake = state["wake"]
    need(type(owner) is int and owner > 1 and type(wake.get("returncode")) is int
         and wake.get("owner_pid") == owner and wake.get("pid") == wake.get("pgid")
         and wake.get("argv") == ["/usr/bin/caffeinate", "-i", "-w", str(owner)], "Unclosed/incorrect wake record")
    pids = {owner, wake["pid"]}; groups = {wake["pgid"]}
    for job in state["jobs"].values():
        need(job.get("owner_pid") == owner and type(job.get("pid")) is int and job["pid"] > 1
             and job.get("pgid") == job["pid"] and type(job.get("returncode")) is int
             and job["returncode"] == 0 and math.isfinite(job["finished_at"]), "Incomplete owned worker")
        pids.add(job["pid"]); groups.add(job["pgid"])
    snapshot = subprocess.check_output(["ps", "-axo", "pid=,pgid=,args="], text=True, timeout=5)
    for line in snapshot.splitlines():
        fields = line.strip().split(None, 2)
        if len(fields) != 3:
            continue
        pid, pgid, command = int(fields[0]), int(fields[1]), fields[2]
        need(pid not in pids and pgid not in groups, "Historical owned process/group is still present")
        if pid == os.getpid():
            continue
        executable = Path(command.split()[0]).name.lower()
        if "python" in executable or "pytest" in executable:
            need(not re.search(r"(?:pytest|sampled[-_]target.*audit|(?:-m\s+scripts\.|scripts/)[\w]*(?:train|eval|study|supervisor|probe|distill|experiment))", command),
                 "Other test/model/audit process is active")

    # Maintain an artifact inventory, not a rebasing inventory of runtime/source files.

    def bind(path, expected=None):
        p = absolute(path)
        need(p.is_relative_to(root) and p.resolve(strict=True) == p and p.is_file(), "Artifact escapes repository: " + str(p))

    def ref(path, expected=None):

    def artifacts(mapping):

            elif p.is_relative_to(root):

            else:

                     and expected_hash in code_hashes and sha(p) == expected_hash,

        need(set(state["receipts"][arm]) == {str(path)} and hex64(expected_hash), "Wrong final receipt binding")
        receipt = read(bind(path, expected_hash)); receipts[arm] = receipt; receipt_hashes[arm] = expected_hash
