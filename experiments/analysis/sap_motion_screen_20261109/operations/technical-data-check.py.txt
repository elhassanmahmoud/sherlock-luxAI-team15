def digest(path):
    h=hashlib.sha256()
    with Path(path).open('rb') as f:
        for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
    return h.hexdigest()
def read(path):return json.loads(Path(path).read_text())

def close(a,b):
    if isinstance(b,dict):return isinstance(a,dict) and set(a)==set(b) and all(close(a[k],v) for k,v in b.items())
    if isinstance(b,list):return isinstance(a,list) and len(a)==len(b) and all(close(x,y) for x,y in zip(a,b))
    if isinstance(b,float):return isinstance(a,(int,float)) and math.isfinite(a) and abs(a-b)<=1e-11
    return a==b

def motion_score(p,sensor,enemy,old,new,drop,counts):
    dx,dy=new[0]-old[0],new[1]-old[1];positions=[];blocked=[];unknown=[]
    for kind,(vx,vy) in enumerate(((dx,dy),(-dy,dx),(-dx,-dy),(dy,-dx),(0,0))):
        t=[old[0]+vx,old[1]+vy]
        if not (0<=t[0]<p['W'] and 0<=t[1]<p['H']) or (sensor[t[0]][t[1]] and not p['passable'][t[0]][t[1]]):t=old.copy();blocked.append(kind)
        elif not sensor[t[0]][t[1]]:unknown.append(kind)
        positions.append(t)
    def weight(c,t):
        d=max(abs(c[0]-t[0]),abs(c[1]-t[1]));return 1. if d==0 else drop if d==1 else 0.
    gain=[weight(new,t)-weight(old,t) for t in positions]
    other=sum(weight(new,t)-weight(old,t) for u,t,en in zip(p['enemy_vis_ids'],p['enemy_coords'],p['enemy_energies']) if u!=enemy and math.isfinite(en) and en>=0)
    n=sum(counts);prob=[(c+1)/(n+5) for c in counts];focus=sum(q*x for q,x in zip(prob,gain))
    variance=max(0.,(sum(q*x*x for q,x in zip(prob,gain))-focus*focus)/(n+6));mean=other+focus
    return dict(sample_count=n,counts=counts.copy(),probabilities=prob,focus_gains=gain,other_gain=other,mean=mean,variance=variance,lower_bound=mean-2*math.sqrt(variance),next_positions=positions,blocked_classes=blocked,unknown_classes=unknown)
