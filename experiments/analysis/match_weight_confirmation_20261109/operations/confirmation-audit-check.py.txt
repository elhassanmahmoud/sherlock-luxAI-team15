def sha(p,fresh=False):
 p=Path(p)
 check(p.is_absolute(),'absolute data path')
 if fresh or str(p) not in cache:
  h=hashlib.sha256()
  with p.open('rb') as f:
   for b in iter(lambda:f.read(1024*1024),b''):h.update(b)
  cache[str(p)]=h.hexdigest()
 return cache[str(p)]
def pin(p,h,label):

def equal(a,b,label):
 if isinstance(a,dict):
  check(isinstance(b,dict) and set(a)==set(b),label+' keys')
  for k in a:equal(a[k],b[k],label+'.'+k)
 elif isinstance(a,list):
  check(isinstance(b,list) and len(a)==len(b),label+' length')
  for i,(x,y) in enumerate(zip(a,b)):equal(x,y,label+f'[{i}]')
 elif isinstance(a,float):check(type(b) in (int,float) and math.isfinite(b) and math.isclose(a,b,abs_tol=EPS,rel_tol=0),label)
 else:check(type(a)==type(b) and a==b,label)
