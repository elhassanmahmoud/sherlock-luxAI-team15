def state_hash(value):
    h=hashlib.sha256()

        if isinstance(v,np.ndarray):
            emit(b'ndarray');emit(v.dtype.str.encode());visit(list(v.shape));emit(v.tobytes(order='C'))
        elif isinstance(v,dict):
            emit(b'dict');emit(str(len(v)).encode())
            for k in sorted(v,key=lambda x:(type(x).__name__,repr(x))):visit(k);visit(v[k])
        elif isinstance(v,(list,tuple)):
            emit(type(v).__name__.encode());emit(str(len(v)).encode())
            for x in v:visit(x)
        elif isinstance(v,np.generic):visit(v.item())
        else:emit(type(v).__name__.encode());emit(repr(v).encode())
    visit(value);return h.hexdigest()
