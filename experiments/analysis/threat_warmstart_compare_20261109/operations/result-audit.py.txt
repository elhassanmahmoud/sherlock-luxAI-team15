def sha(p):

def bootstrap(values):

def equal(a,b,label):
 if isinstance(a,dict):
  check(isinstance(b,dict) and set(a)==set(b),label+' keys')
  for k in a:equal(a[k],b[k],label+'.'+k)
 elif isinstance(a,list):
  check(isinstance(b,list) and len(a)==len(b),label+' length')
  for i,(x,y) in enumerate(zip(a,b)):equal(x,y,label+f'[{i}]')
 elif isinstance(a,float):check(isinstance(b,(int,float)) and math.isfinite(b) and math.isclose(a,b,abs_tol=1e-12,rel_tol=0),label)
 else:check(type(a)==type(b) and a==b,label)
def number(row,k,integer=False):
