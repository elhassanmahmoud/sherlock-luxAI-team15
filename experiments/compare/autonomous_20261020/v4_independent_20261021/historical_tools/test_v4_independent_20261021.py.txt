class Tests(unittest.TestCase):
    def test_fixed_design_and_commands(self):
        design = M.design()
        self.assertEqual([r[0] for r in design], ['direct_v4_vs_v3', 'fresh_v3', 'fresh_v4'])
        self.assertEqual(sum(count * 2 * len(opps) for _, _, _, count, opps in design), 2400)
        self.assertEqual([(s, n) for _, _, s, n, _ in design], [(510400, 400), (510800, 200), (510800, 200)])
        for i, (label, weight, first, count, opponents) in enumerate(design):
            command = M.D.eval_command(weight, M.OUT / f'{label}.csv', first=first - 500000, n=count, opponents=','.join(opponents))
            self.assertEqual(command[0], str(M.ROOT / '.venv/bin/python'))
            self.assertEqual(command[command.index('--first') + 1], '10400' if i == 0 else '10800')
            self.assertEqual(command[command.index('--n') + 1], '400' if i == 0 else '200')
            self.assertEqual(command[command.index('--workers') + 1], '2')
            self.assertNotIn('--set', command)
            self.assertEqual(opponents, ('self:' + str(M.V3),) if i == 0 else ('default', 'solution92'))

    def test_both_benchmarks_run_even_after_losing_direct_result(self):
        calls = []
        def fake_job(label, *args):
            calls.append(label)
            return {'all_scores': 0} if label.startswith('direct') else {'fixture': 1}
        with patch.object(M.D, 'STATE', {'jobs': []}), patch.object(M.D, 'save_state'), patch.object(M.C, 'run_job', side_effect=fake_job), patch.object(M, 'decide', side_effect=AssertionError('premature gate')):
            tables = M.execute_all_jobs()
        self.assertEqual(calls, [r[0] for r in M.design()])
        self.assertEqual(len(tables), 3)

    def test_gate_boundaries_and_no_promotion(self):
        comparison = {'solution92': {'half_units': 1, 'ci95': [.00001, .1]}, 'default': {'half_units': -16}}
        direct = {'ci95': [.50001, .8]}
        self.assertEqual(M.decide(comparison, direct)['decision'], 'eligible_for_technical_review')
        self.assertFalse(M.decide(comparison, direct)['promotion_authorized'])
        comparison['default']['half_units'] = -17
        self.assertEqual(M.decide(comparison, direct)['decision'], 'no_promotion_gate_failed')
        comparison['default']['half_units'] = -16
        direct['ci95'][0] = .5
        self.assertEqual(M.decide(comparison, direct)['decision'], 'no_promotion_gate_failed')
        direct['ci95'][0] = .50001
        comparison['solution92']['ci95'][0] = 0
        self.assertEqual(M.decide(comparison, direct)['decision'], 'no_promotion_gate_failed')
        comparison['solution92']['ci95'][0] = .01
        comparison['solution92']['half_units'] = 0
        self.assertEqual(M.decide(comparison, direct)['decision'], 'no_promotion_gate_failed')

    def test_bootstrap_extremes_and_pairing(self):
        for n in (200, 400):
            for v in (0., .5, 1.):
                stat = M.C.bootstrap([v] * n)
                self.assertEqual(stat['mean'], v)
                self.assertEqual(stat['ci95'], [v, v])
                self.assertEqual((stat['maps'], stat['bootstrap_samples'], stat['bootstrap_seed']), (n, 5000, 0))
        for bad in ([], [float('nan')], [float('inf')], [[0, 1]]):
            with self.assertRaises(RuntimeError):
                M.C.bootstrap(bad)
        opponent = 'self:' + str(M.V3)
        direct = {(str(s), side, opponent): {'result': str(float(i))}
                  for s in range(510400, 510800) for i, side in enumerate(M.SIDES)}
        stat = M.direct_stat(direct)
        self.assertEqual(stat['ci95'], [.5, .5])  # independent sides would create spurious uncertainty
        self.assertEqual((stat['maps'], stat['games']), (400, 800))
        self.assertFalse(M.C.direct_gate(stat)['passed'])
        for i, s in enumerate(range(510400, 510800)):
            for side in M.SIDES:
                direct[(str(s), side, opponent)] = {'result': str(float(i < 300))}
        stat = M.direct_stat(direct)
        values = np.array([1.] * 300 + [0.] * 100)
        ix = np.random.default_rng(0).integers(0, 400, size=(5000, 400))
        np.testing.assert_array_equal(stat['ci95'], np.percentile(values[ix].mean(1), [2.5, 97.5]))

    def test_fresh_paired_delta(self):
        reference, candidate = {}, {}
        for i, s in enumerate(range(510800, 511000)):
            for j, side in enumerate(M.SIDES):
                for opp in M.BENCHMARKS:
                    key = (str(s), side, opp)
                    reference[key] = {'result': '.5'}
                    candidate[key] = {'result': str(float(j) if opp == 'default' else float(i < 120))}
        result = M.fresh_comparison(reference, candidate)
        self.assertEqual(result, M.fresh_comparison(reference, candidate))
        self.assertEqual(result['default']['ci95'], [0., 0.])
        values = np.array([.5] * 120 + [-.5] * 80)
        ix = np.random.default_rng(0).integers(0, 200, size=(5000, 200))
        np.testing.assert_array_equal(result['solution92']['ci95'], np.percentile(values[ix].mean(1), [2.5, 97.5]))
        self.assertEqual(result['solution92']['half_units'], 80)
        for record in result.values():
            self.assertEqual((record['maps'], record['games_per_arm']), (200, 400))
            self.assertAlmostEqual(record['delta'], record['half_units'] / 800)
            self.assertAlmostEqual(record['candidate_mean'] - record['reference_mean'], record['delta'])

    def fixture(self, first, count, opponents):
        fields = M.literal_assignment(M.ROOT / 'scripts/experiment.py', 'GAME_FIELDS')
        stream = io.StringIO()
        writer = csv.DictWriter(stream, fieldnames=fields)
        writer.writeheader()
        for seed in range(first, first + count):
            for side in M.SIDES:
                for opp in opponents:
                    row = dict.fromkeys(fields, 0)
                    row.update(seed=seed, side=side, opponent=opp, steps=303, point_counting_version=2,
                               our_wins=2, opp_wins=1, result=1, game_win=1, point_win=1, our_points=1)
                    writer.writerow(row)
        return stream.getvalue()

    def test_strict_csv_validation(self):
        for _, _, first, count, opponents in M.design():
            valid = self.fixture(first, count, opponents)
            with tempfile.TemporaryDirectory(prefix='v4-runner-test-') as td:
                path = Path(td) / 'data.csv'
                path.write_text(valid)
                self.assertEqual(len(M.read_rows(path, first, count, opponents)), 800)
                lines = valid.splitlines()
                fields = next(csv.reader([lines[0]]))
                bad = ['\n'.join(lines[:-1]), valid + lines[-1] + '\n', valid.replace(str(first), str(first-1))]
                for field, value in [('steps', '302'), ('point_counting_version', '1'), ('secs', 'nan'), ('result', '.5'), ('our_wins', '1'), ('our_points', '-1'), ('point_win', '.5'), ('game_win', '0')]:
                    row = next(csv.reader([lines[1]])); row[fields.index(field)] = value
                    stream = io.StringIO(); writer = csv.writer(stream); writer.writerow(row)
                    bad.append('\n'.join([lines[0], stream.getvalue().strip(), *lines[2:]]) + '\n')
                bad.extend([valid.replace('secs', 'unexpected'), valid.replace('secs', 'steps')])
                for data in bad:
                    path.write_text(data)
                    with self.assertRaises((RuntimeError, ValueError, KeyError)):
                        M.read_rows(path, first, count, opponents)
